/* ========================= ✅ 메인 홍보 슬라이드 팝업 - 이미지+텍스트 혼합 / 쿠팡 슬라이드 느낌 - "오늘 하루 보지 않기" 지원 ========================= */ const promoModal = document.getElementById("promoModal"); const promoTrack = document.getElementById("promoTrack"); const promoDots = document.getElementById("promoDots"); const promoNoShowToday = document.getElementById("promoNoShowToday"); /* ✅ 홍보 슬라이드 데이터 (이미지 파일만 교체하면 됨) - repo 루트에 promo1.jpg / promo2.jpg / promo3.jpg 업로드 추천 */ const PROMO_SLIDES = [ { tag:"인터넷TV", right:"현금지원", title:"인터넷TV 가입시 최대 현금지원", sub:"카톡에 “지역/통신사/희망상품”만 보내면 바로 안내", img:"promo1.jpg" }, { tag:"특가폰", right:"준공표", title:"특가 + 준공표 · 바로 개통", sub:"새제품 풀박스 / 최저가 보장 (조건 문의)", img:"promo2.jpg" }, { tag:"긴급", right:"마감임박", title:"재고소진 임박 · D-1", sub:"지금이 기회! 마감 직전 특가 (선착순)", img:"promo3.jpg" } ]; /* ✅ 오늘 하루 숨김 키 */ function getPromoBlockKey(){ const parts = new Intl.DateTimeFormat("en-CA", { timeZone:"Asia/Seoul", year:"numeric", month:"2-digit", day:"2-digit" }).formatToParts(new Date()); const get = (t) => (parts.find(p => p.type === t)?.value || ""); const y = get("year"), m = get("month"), d = get("day"); return `PROMO_BLOCK_${y}-${m}-${d}`; } function buildPromoSlides(){ if(!promoTrack || !promoDots) return; promoTrack.innerHTML = ""; promoDots.innerHTML = ""; PROMO_SLIDES.forEach((s, idx)=>{ promoTrack.insertAdjacentHTML("beforeend", `
${escapeHtml(s.tag)}
${escapeHtml(s.right)}
${escapeHtml(s.title)}
${escapeHtml(s.sub)}
`); promoDots.insertAdjacentHTML("beforeend", `
`); }); const items = Array.from(promoTrack.querySelectorAll(".promoItem")); const dots = Array.from(promoDots.querySelectorAll(".promoDot")); function setActive(i){ dots.forEach((d, idx)=> d.classList.toggle("active", idx===i)); } promoTrack.addEventListener("scroll", ()=>{ let bestIdx = 0, bestDist = Infinity; items.forEach((el, idx)=>{ const rect = el.getBoundingClientRect(); const cx = rect.left + rect.width/2; const cRect = promoTrack.getBoundingClientRect(); const cc = cRect.left + cRect.width/2; const dist = Math.abs(cx - cc); if(dist < bestDist){ bestDist = dist; bestIdx = idx; } }); setActive(bestIdx); }, { passive:true }); /* ✅ 자동 넘김(쿠팡 느낌) - 팝업 열려있을 때만 */ let timer = null; function startAuto(){ stopAuto(); timer = setInterval(()=>{ if(!promoModal.classList.contains("show")) return; const cRect = promoTrack.getBoundingClientRect(); const cc = cRect.left + cRect.width/2; let cur = 0, best = Infinity; items.forEach((el, idx)=>{ const r = el.getBoundingClientRect(); const cx = r.left + r.width/2; const dist = Math.abs(cx - cc); if(dist < best){ best = dist; cur = idx; } }); const next = (cur + 1) % items.length; items[next].scrollIntoView({ behavior:"smooth", inline:"center", block:"nearest" }); }, 3500); } function stopAuto(){ if(timer){ clearInterval(timer); timer = null; } } promoModal.addEventListener("transitionend", ()=>{ if(promoModal.classList.contains("show")) startAuto(); else stopAuto(); }); } /* ✅ 표시/닫기 로직 */ function shouldShowPromo(){ return String(StorageSafe.get(getPromoBlockKey()) ?? "") !== "1"; } function showPromo(){ if(!promoModal) return; promoNoShowToday.checked = false; showModal(promoModal); } function closePromo(){ if(promoNoShowToday.checked){ StorageSafe.set(getPromoBlockKey(), "1"); } hideModal(promoModal); } document.getElementById("promoCloseBtn")?.addEventListener("click", closePromo); document.getElementById("promoOkBtn")?.addEventListener("click", closePromo); promoNoShowToday?.addEventListener("change", (e)=>{ if(e.target.checked) StorageSafe.set(getPromoBlockKey(), "1"); else StorageSafe.remove(getPromoBlockKey()); });
